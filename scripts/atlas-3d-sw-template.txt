const VERSION='__VERSION__';
const FILES=__FILES__;
const EXTRAS=__EXTRAS__;
const CACHE='tariklab-atlas-3d-'+VERSION;
const EXTRA_CACHE='tariklab-atlas-3d-systems-'+VERSION;
const ROOT='/atlas/3d/';
const pathFor=file=>file.url||ROOT+file.path;
const digest=async bytes=>Array.from(new Uint8Array(await crypto.subtle.digest('SHA-256',bytes)),b=>b.toString(16).padStart(2,'0')).join('');
async function valid(response,file){if(!response||response.status!==200)return false;const bytes=await response.clone().arrayBuffer();return bytes.byteLength===file.bytes&&await digest(bytes)===file.sha256;}
self.addEventListener('install',event=>event.waitUntil((async()=>{try{const cache=await caches.open(CACHE);for(const file of FILES){const path=pathFor(file),response=await fetch(path,{cache:'reload'});if(!await valid(response,file))throw new Error('Incomplete or corrupted Atlas 3D package');const safe=new Response(await response.arrayBuffer(),{status:200,headers:response.headers});await cache.put(path,safe);}}catch(error){await caches.delete(CACHE);throw error;}})()));
self.addEventListener('activate',event=>event.waitUntil(self.clients.claim()));
// No forced activation or deletion of other versions: open tabs keep their cohort.
self.addEventListener('fetch',event=>{const url=new URL(event.request.url);if(event.request.method!=='GET'||url.origin!==location.origin)return;const inside=url.pathname.startsWith(ROOT);let relative=inside?url.pathname.slice(ROOT.length):'';if(inside&&!relative)relative='index.html';const file=FILES.find(f=>f.url?f.url===url.pathname:inside&&f.path===relative);const extra=inside?EXTRAS.find(f=>ROOT+f.path===url.pathname):null;if(!file&&!extra)return;event.respondWith((async()=>{const record=file||extra;const cache=await caches.open(file?CACHE:EXTRA_CACHE),path=pathFor(record),response=await cache.match(path);if(await valid(response,record))return response;const fresh=await fetch(event.request);if(!await valid(fresh,record))throw new Error('Atlas 3D cohort mismatch');const safe=new Response(await fresh.arrayBuffer(),{status:200,headers:fresh.headers});await cache.put(path,safe.clone());return safe;})());});
self.addEventListener('message',event=>{if(event.data==='verify'){event.waitUntil((async()=>{const cache=await caches.open(CACHE);let ready=true;for(const file of FILES)if(!await valid(await cache.match(pathFor(file)),file)){ready=false;break;}event.ports[0]?.postMessage({ready,version:VERSION});})());return;}if(event.data?.type!=='cache-system')return;event.waitUntil((async()=>{const port=event.ports[0];try{const {sex,systems}=event.data;if(!['male','female'].includes(sex)||!Array.isArray(systems)||!systems.every(s=>typeof s==='string'))throw Error('Invalid system selection');const selected=EXTRAS.filter(f=>f.sex===sex&&systems.includes(f.system));const cache=await caches.open(EXTRA_CACHE);for(const file of selected){const path=pathFor(file);if(await valid(await cache.match(path),file))continue;const fresh=await fetch(path,{cache:'reload'});if(!await valid(fresh,file))throw Error('System asset integrity failed');await cache.put(path,new Response(await fresh.arrayBuffer(),{status:200,headers:fresh.headers}));}port?.postMessage({ready:true,files:selected.length});}catch(error){port?.postMessage({ready:false,error:error.message});}})());});
